# syzkaller descriptions of the MoobFS layer (security roadmap S3, journal 0109). Copied into
# sys/linux/ of a syzkaller checkout by vm/syzkaller-local.sh. Layouts follow the headers
# moobfs_ioctl.h and moobfs_attr.h of the repository; the image (vm/syzkaller/image-setup.sh)
# mounts the layer at /moob/layer over /moob/lower, with domains d0..d3 and typed files in them.

include <uapi/linux/moobfs_ioctl.h>
include <uapi/linux/moobfs_attr.h>
include <uapi/linux/fcntl.h>

# directories of the layer, usable wherever a directory descriptor is (getdents64, fchdir)
resource fd_moobfs_root[fd_dir]
resource fd_moobfs_dom[fd_dir]
# a file opened inside the layer: the attribute calls take it, so they reach the layer's xattr
# handlers instead of whatever descriptor the program holds (journal 0138: moobl_xattr_get was
# never reached with a plain fd); still usable wherever any fd is
resource fd_moobfs_file[fd]

# The executor runs each program in a private root, so the program mounts the layer itself: a
# directory it created as the lower (relative paths resolve from its working directory), another
# as the mount point. Files, attributes, renames and queries then go through the layer.
mount$moobfs_layer(src const[0], dst ptr[in, filename], type ptr[in, string["moobfs_layer"]], flags flags[mount_flags], opts ptr[in, moobfs_mount_data])
umount2$moobfs(path ptr[in, filename], flags flags[umount_flags])

moobfs_mount_data {
	lower	stringnoz[moobfs_lowers]
	opts	fs_options[moobfs_options]
} [packed]

moobfs_lowers = "lower=./file0,", "lower=./file1,", "lower=./file0/file1,", "lower=.,", "lower=./file2,"

moobfs_options [
	nolinks		stringnoz["nolinks"]
	noindex		stringnoz["noindex"]
	budget		fs_opt_dec_suffix["index_budget"]
	depth		fs_opt_dec["domain_depth", int32[0:10]]
	idle		fs_opt_dec["index_idle", int32]
	lower		fs_opt_str["lower"]
] [varlen]

# a layer's root (QUERY, GET_INFO) and its domain directories (QUERY2), by any name
openat$moobfs_root(fd const[AT_FDCWD], file ptr[in, filename], flags const[O_RDONLY], mode const[0]) fd_moobfs_root
openat$moobfs_dom(fd const[AT_FDCWD], file ptr[in, filename], flags const[O_RDONLY], mode const[0]) fd_moobfs_dom
openat$moobfs_file(fd const[AT_FDCWD], file ptr[in, filename], flags flags[open_flags], mode flags[open_mode]) fd_moobfs_file

ioctl$MOOB_IOC_QUERY(fd fd_moobfs_root, cmd const[MOOB_IOC_QUERY], arg ptr[inout, moob_query])
ioctl$MOOB_IOC_GET_INFO(fd fd_moobfs_root, cmd const[MOOB_IOC_GET_INFO], arg ptr[out, moob_info])
ioctl$MOOB_IOC_QUERY2(fd fd_moobfs_dom, cmd const[MOOB_IOC_QUERY2], arg ptr[inout, moob_query2_t[moob_cursor_out]])

ioctl$MOOB_IOC_QUERY2_BATCH(fd fd_moobfs_dom, cmd const[MOOB_IOC_QUERY2_BATCH], arg ptr[in, moob_query2_batch])

# A few values of each type, so that two files or a file and a term often hold equal, ordered or
# special values (value_cmp, journal 0138: 12 of 24 PCs with random 8-byte values). Strings
# share prefixes; the two long ones are longer than the cursor's 96 inline bytes and share them.
# Doubles are bit patterns: 0, -0, 1, -1, 2, +inf, -inf, quiet and signalling NaN (refused),
# the smallest denormal and the largest finite value.
moob_str_vals = "note", "nota", "notes", "not", "task", "x", "y", "", "aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa", "aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaab"
moob_str_lens = 0, 1, 3, 4, 5, 100, 1024, 1025
moob_int_vals = 0, 1, 2, 3, 7, 0x10, 0x20, 0xffffffffffffffff, 0xfffffffffffffffb, 0x7fffffffffffffff, 0x8000000000000000
moob_dbl_vals = 0x0, 0x8000000000000000, 0x3ff0000000000000, 0xbff0000000000000, 0x4000000000000000, 0x7ff0000000000000, 0xfff0000000000000, 0x7ff8000000000000, 0xfff8000000000000, 0x7ff0000000000001, 0x1, 0x7fefffffffffffff

moob_value [
	str	moob_val_str
	strv	moob_val_strv
	i64	moob_val_8[MOOB_T_INT64, flags[moob_int_vals, int64]]
	dbl	moob_val_8[MOOB_T_DOUBLE, flags[moob_dbl_vals, int64]]
	tm	moob_val_8[MOOB_T_TIME, flags[moob_int_vals, int64]]
	i64r	moob_val_8[MOOB_T_INT64, int64]
	dblr	moob_val_8[MOOB_T_DOUBLE, int64]
	bool	moob_val_bool
	raw	moob_val_raw
	junk	array[int8, 0:16]
] [varlen]

moob_val_str {

} [packed]

moob_val_strv {

	type	const[T, int8]
	v	V
} [packed]

moob_val_bool {
	type	const[MOOB_T_BOOL, int8]
	v	int8[0:2]
} [packed]

moob_val_raw {
	type	const[MOOB_T_RAW, int8]

# Layouts of moobfs_ioctl.h with natural alignment, as the C compiler lays them out (sizes checked:
# term 1098, query 4440, query2 4656, info 104, batch 32, result 272; cursor at offset 104 of
# query2, terms at 264, value at 70 of a term)
moob_ops = 1, 2, 3, 4, 5, 6, 7
moob_limits = 0, 1, 2, 4, 4096, 4097

# A term is one of: a value of each queryable type with a valid length and a chosen value, an
# EXISTS term, or any bytes (the invalid shapes term_check refuses). The typed ones are packed
# with the value at offset 70, as the C layout puts value[] there, and padded with zeros to 1098.
moob_query_term [
	str	moob_term_t[MOOB_T_STRING, flags[moob_str_lens, int16], string[moob_str_vals, 1024]]
	i64	moob_term_t[MOOB_T_INT64, const[8, int16], flags[moob_int_vals, int64]]
	tm	moob_term_t[MOOB_T_TIME, const[8, int16], flags[moob_int_vals, int64]]
	dbl	moob_term_t[MOOB_T_DOUBLE, const[8, int16], flags[moob_dbl_vals, int64]]
	bool	moob_term_t[MOOB_T_BOOL, const[1, int16], int8[0:2]]
	exists	moob_term_exists
	any	moob_term_any
]

type moob_term_t[T, L, V] {
	name	string[moob_attr_names, 65]
	op	flags[moob_ops, int8]
	type	const[T, int8]
	flags	int8[0:1]
	len	L
	value	V
} [packed, size[1098]]

moob_term_exists {
	name	string[moob_attr_names, 65]
	op	const[7, int8]
	type	const[0, int8]
	flags	const[0, int8]
	len	const[0, int16]
} [packed, size[1098]]

moob_term_any {
	name	string[moob_attr_names, 65]
	op	int8[0:8]
	type	int8[0:7]
	flags	int8[0:2]
	len	int16[0:1100]
	value	array[int8, 1024]
	pad	array[int8, 3]
}

moob_query_result {
	ino		int64
	flags		int32
	path_len	int32
	path		array[int8, 256]
}

moob_query {
	version		int32[0:2]
	nterms		int32[0:5]
	limit		flags[moob_limits, int32]
	flags		int32[0:1]
	results		ptr64[out, array[moob_query_result]]

moob_info {

# The QUERY2 cursor is 128 bytes under a MAC keyed per mount, so random bytes never continue a
# query (journal 0138: query2_exec at 41 of 75 PCs). Each of its 16 little-endian words is a
# resource of its own. ioctl$MOOB_IOC_QUERY2 (and each request of a batch) leaves the cursor
# unwritten (zero, a new query, unless an earlier call of the program wrote that memory) and
# returns its words;
# ioctl$MOOB_IOC_QUERY2_NEXT sends words returned earlier, word for word, to continue that
# query (or, mixed from different queries, to be refused with ESTALE) and returns the next ones.
# A resource cannot be optional and still be created (pkg/compiler), hence the two calls.
resource moob_cw0[int64]: 0
resource moob_cw1[int64]: 0
resource moob_cw2[int64]: 0
resource moob_cw3[int64]: 0
resource moob_cw4[int64]: 0
resource moob_cw5[int64]: 0
resource moob_cw6[int64]: 0
resource moob_cw7[int64]: 0
resource moob_cw8[int64]: 0
resource moob_cw9[int64]: 0
resource moob_cw10[int64]: 0
resource moob_cw11[int64]: 0
resource moob_cw12[int64]: 0
resource moob_cw13[int64]: 0
resource moob_cw14[int64]: 0
resource moob_cw15[int64]: 0

moob_cursor_out {

	w5	moob_cw5	(out)
	w6	moob_cw6	(out)
	w7	moob_cw7	(out)
	w8	moob_cw8	(out)
	w9	moob_cw9	(out)

	w12	moob_cw12	(out)
